$('#ballotingempire').on('click', function(e) {

    e.preventDefault();

    $.ajax({
        url: "https://sts.ug.edu.gh/services/empire-balloting",
        type: "GET",
        dataType: "json",

        beforeSend: function() {
            // Optional loading indicator
        },

        success: function(res) {

            if (res.status === 'success') {

                var data = res.data;

                /*
                 * Reset form first
                 */
                $('#ballotformempire')[0].reset();

                /*
                 * Clear all iCheck checkboxes
                 */
                $('.room_category').iCheck('uncheck');

                /*
                 * No previous submission
                 */
                if (!data) {

                    $('#ballotingresponseempire').val('');
                    $('#phoneballotingempire').val('');

                    $('#showroomcategory').hide();

                    $('#ballotmodalempire').modal('show');

                    return;
                }

                /*
                 * Populate response
                 */
                $('#ballotingresponseempire').val(data.response);

                /*
                 * Populate phone
                 */
                $('#phoneballotingempire').val(data.phone);

                /*
                 * Populate room categories
                 *
                 * Database value:
                 * 2in1A,2in1B,4in1A
                 */
                if (data.room_category) {

                    var categories = data.room_category.split(',');

                    $.each(categories, function(index, category) {

                        category = $.trim(category);

                        $('.room_category[value="' + category + '"]')
                            .iCheck('check');

                    });
                }

                /*
                 * Show/hide room category depending
                 * on response
                 */
                if (data.response === 'Yes') {
                    $('#showroomcategory').show();
                } else {
                    $('#showroomcategory').hide();
                }

                /*
                 * Update the response displayed
                 * on the main card
                 */
                updateEmpireResponse(data);

                /*
                 * Open modal
                 */
                $('#ballotmodalempire').modal('show');

            } else {

                swal(
                    'Notice',
                    'You are not eligible for the Empire Hostel random allocation exercise.',
                    'warning'
                );

            }

        },

        error: function(xhr) {

            console.log(xhr.responseText);

            swal(
                'Error',
                'Unable to load your balloting information.',
                'error'
            );

        }

    });

});

$('#ballotformempire').on('submit', function(e) {

    e.preventDefault();

    var response = $('#ballotingresponseempire').val();

    /*
     * Validate response
     */
    if (!response) {

        swal(
            'Required',
            'Please select whether you would like to participate.',
            'warning'
        );

        return;
    }

    /*
     * If Yes, require at least one room category
     */
    if (response === 'Yes') {

        var selectedCategories = [];

        $('.room_category:checked').each(function() {
            selectedCategories.push($(this).val());
        });

        if (selectedCategories.length === 0) {

            swal(
                'Required',
                'Please select at least one room category.',
                'warning'
            );

            return;
        }
    }

    /*
     * Disable submit button
     */
    $('#submitbtn_ballot_empire').prop('disabled', true);

    $('#loadinggif255').show();

    $.ajax({

        url: "https://sts.ug.edu.gh/services/empire-save-balloting",

        type: "POST",

        data: $(this).serialize(),

        dataType: "json",

        success: function(res) {

            if (res.status === 'success') {

                var data = res.data;

                /*
                 * Update modal with saved information
                 */
                $('#ballotingresponseempire').val(data.response);

                $('#phoneballotingempire').val(data.phone);

                /*
                 * Update room categories
                 */
                $('.room_category').iCheck('uncheck');

                if (data.room_category) {

                    var categories = data.room_category.split(',');

                    $.each(categories, function(index, category) {

                        category = $.trim(category);

                        $('.room_category[value="' + category + '"]')
                            .iCheck('check');

                    });
                }

                /*
                 * Update the H5 on the main card
                 */
                updateEmpireResponse(data);

                /*
                 * Close modal
                 */
                $('#ballotmodalempire').modal('hide');

                swal(
                    'Saved',
                    'Your Empire Hostel balloting response has been saved successfully.',
                    'success'
                );

            } else {

                swal(
                    'Error',
                    'Your response could not be saved.',
                    'error'
                );
            }

        },

        error: function(xhr) {

            console.log(xhr.responseText);

            swal(
                'Error',
                'An error occurred while saving your response.',
                'error'
            );

        },

        complete: function() {

            $('#submitbtn_ballot_empire').prop('disabled', false);

            $('#loadinggif255').hide();

        }

    });

});

function updateEmpireResponse(data) {

    var html = '';

    if (data.response === 'Yes') {

        html += '<strong>Response: YES</strong>';

        if (data.room_category) {

            var categories = data.room_category.split(',');

            html += '<br>';

            html += '<small>Room Categories: ';

            $.each(categories, function(index, category) {

                category = $.trim(category);

                if (index > 0) {
                    html += ', ';
                }

                html += getRoomCategoryName(category);
            });

            html += '</small>';
        }

    } else {

        html = '<strong>Response: NO</strong>';

    }

    $('#responseempire').html(html);
}

function getRoomCategoryName(category) {

    var names = {

        '2in1A': '2 in 1 AC Type A',

        '2in1B': '2 in 1 AC Type B',

        '4in1A': '4 in 1 No AC Type A',

        '4in1B': '4 in 1 No AC Type B'

    };

    return names[category] || category;
}

    function showAlert($el, message, type) {
        $el.removeClass('d-none alert-success alert-danger alert-info alert-warning')
            .addClass('alert-' + type)
            .text(message);
    }

    function hideAlert($el) {
        $el.addClass('d-none').text('');
    }

    function badgeHtml(value, trueLabel, falseLabel) {
        if (parseInt(value, 10) === 1) {
            return '<span class="badge badge-success">' + trueLabel + '</span>';
        }
        return '<span class="badge badge-secondary">' + falseLabel + '</span>';
    }

    function hasUnverifiedPhones(phones) {
        return phones.some(function (phone) {
            return parseInt(phone.verified, 10) !== 1 && parseInt(phone.active, 10) === 1;
        });
    }

    function canCloseModal(phones) {
        if (!phones.length) {
            return true;
        }

        return phones.every(function (phone) {
            return parseInt(phone.verified, 10) === 1 || parseInt(phone.active, 10) === 0;
        });
    }

    function updateCloseButton(phones) {
        var $btn = $('#closePhoneVerificationModalBtn');
        $btn.prop('disabled', false);
        return
        if (canCloseModal(phones)) {
            $btn.prop('disabled', false);
        } else {
            $btn.prop('disabled', true);
        }
    }

    function renderPhoneTable(phones) {
        var $tbody = $('#phoneVerificationTableBody');
        $tbody.empty();

        if (!phones.length) {
            $tbody.append(
                '<tr><td colspan="4" class="text-center text-muted">No phone numbers found. Add one below.</td></tr>'
            );
            updateCloseButton(phones);
            return;
        }

        phones.forEach(function (phone) {
            var verified = parseInt(phone.verified, 10) === 1;
            var active = parseInt(phone.active, 10) === 1;
            var actions = '';

            if (!verified && active) {
                actions =
                    '<button type="button" class="btn btn-success mr-1 confirm-phone-btn" data-id="' + phone.id + '" data-answer="yes">' +
                    '<i class="fa fa-check"></i> Yes, Mine</button>' +
                    '<button type="button" class="btn btn-danger confirm-phone-btn" data-id="' + phone.id + '" data-answer="no">' +
                    '<i class="fa fa-times"></i> Not Mine</button>';
            } else if (verified) {
                actions = '<span class="text-success"><i class="fa fa-check-circle"></i> Confirmed</span>';
            } else {
                actions = '<span class="text-muted">Inactive</span>';
            }

            $tbody.append(
                '<tr data-phone-id="' + phone.id + '">' +
                '<td>' + phone.phone + '</td>' +
                '<td>' + actions + '</td>' +
                '<td>' + badgeHtml(phone.verified, 'Yes', 'No') + '</td>' +
                '<td>' + badgeHtml(phone.active, 'Yes', 'No') + '</td>' +
               
                '</tr>'
            );
        });

                }

        if (resendCooldownTimer) {
            clearInterval(resendCooldownTimer);
            resendCooldownTimer = null;
        }
    }

    function startResendCooldown(seconds) {
        var $btn = $('#resendOtpBtn');
        var $countdown = $('#resendOtpCountdown');
        var remaining = seconds || resendCooldownSeconds;

        clearResendCooldown();
        $btn.prop('disabled', true);
        $countdown.text(remaining);

        resendCooldownTimer = setInterval(function () {
            remaining -= 1;
            $countdown.text(remaining);

            if (remaining <= 0) {
                clearResendCooldown();
                if (retriesLeft > 0) {
                    $btn.prop('disabled', false);
                }
                $countdown.text('0');
            }
